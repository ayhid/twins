pub enum Stage { Walk, SizeGrouping, PartialHash, FullHash, Verify } // #[non_exhaustive]; Display "walk", "size grouping", "partial hash", "full hash", "verify"
pub enum Event { StageStarted { stage: Stage, step: u8, steps: u8 }, Progress { stage: Stage, done: u64, total: Option<u64> },

pub trait Observer: Send + Sync { fn on_event(&self, event: &Event); }
pub fn scan(spec: &ScanSpec, observer: &dyn Observer, cancel: &CancelToken) -> Result<ScanOutcome, PipelineError>;
